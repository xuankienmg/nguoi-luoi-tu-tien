**Goal:** Thêm tab con **Lò Rèn** trong Luyện chế: rèn từng ô trang bị lên +0…+99, mỗi nhát rút ngẫu nhiên 5 kết quả (Đại thành / Thành công / Bất biến / Tổn hại / Vỡ nát), có thanh Hỏa Khí → Thần Hỏa chống vỡ, Hồi Thiên Phù hoàn nhát vừa rèn, nhật ký rèn.

**Architecture:** Cấp rèn gắn với **ô trang bị** (giống Tinh Thạch khảm theo ô), không gắn với món đồ — vì túi đồ chỉ lưu `ItemId + Count`. Luật thuần (tỉ lệ, áp kết quả, chi phí, % chỉ số) nằm trong `Game/ForgeData.cs` để test không cần DB. Hành động nằm trong `Game/GameService.Forge.cs`; server roll ngẫu nhiên, client chỉ hiển thị. Lưu 2 cột trên `Player`: `ForgeLevels` (chuỗi gọn `slot:cấp;…`, đọc mỗi lần tính `Stats`) và `ForgeState` (JSON: Hỏa Khí, Thần Hỏa, nhát có thể hoàn, nhật ký 50 dòng).

**Tech Stack:** ASP.NET Core 8 (C#), EF Core (SQLite offline + PostgreSQL online), xUnit (`test/GemSystemTests`), JS thuần trong `wwwroot/`, `node --test`.

```csharp
using System.Globalization;
using System.Text.Json;
using VanDao.Data;

namespace VanDao.Game;

/// <summary>Lò Rèn: cường hóa theo ô trang bị (+0…+15). Cấp gắn với ô, không với món: thay đồ không mất cấp.
/// Player.ForgeLevels = "slot:cấp;…" (đọc trong Stats), Player.ForgeState = JSON ForgeSave (Hỏa Khí, nhát có thể hoàn, nhật ký).</summary>
public static class ForgeData
{
    public const string Iron = "tinh_thiet", Rewind = "hoi_thien_phu";
    public const int MaxLevel = 15, FrenzyCharge = 10, FrenzyStrikes = 3, HistoryKeep = 50;
    // cân bằng 11/2026: theo bản thiết kế Lò Rèn
    public const double Great = 0.08, Hold = 0.15, Damage = 0.12;
    public const double BreakBase = 0.127, BreakPerLevel = 0.034, BreakMax = 0.60;

    public enum Outcome { Great, Success, Hold, Damage, Break }
    public static readonly Outcome[] Order = [Outcome.Great, Outcome.Success, Outcome.Hold, Outcome.Damage, Outcome.Break];
    public static string Key(Outcome o) => o switch
    {
        Outcome.Great => "great", Outcome.Success => "success", Outcome.Hold => "hold", Outcome.Damage => "damage", _ => "break",
    };

    public record Odds(double Great, double Success, double Hold, double Damage, double Break)
    {
        public double Of(Outcome o) => o switch
        {
            Outcome.Great => Great, Outcome.Success => Success, Outcome.Hold => Hold, Outcome.Damage => Damage, _ => Break,
        };
        public double[] ToArray() => Order.Select(Of).ToArray();
    }

    /// <summary>Tỉ lệ nhát kế tiếp. shielded = Thần Hỏa đang cháy (không thể vỡ).</summary>
    public static Odds OddsAt(int level, bool shielded)
    {
        var brk = shielded || level <= 0 ? 0 : Math.Min(BreakMax, BreakBase + BreakPerLevel * level);
        var dmg = level <= 0 ? 0 : Damage;
        return new(Great, Math.Max(0, 1 - brk - Great - dmg - Hold), Hold, dmg, brk);
    }

    /// <summary>r ∈ [0,1): đi lần lượt Order; sai số làm tròn rơi vào kết quả cuối còn tỉ lệ > 0.</summary>
    public static Outcome Roll(Odds o, double r)
    {
        var acc = 0.0;
        foreach (var k in Order) { acc += o.Of(k); if (r < acc) return k; }
        return Order.Last(k => o.Of(k) > 0);
    }

    public sealed class SlotState
    {
        public int L { get; set; }   // cấp
        public int C { get; set; }   // Hỏa Khí 0..9
        public int F { get; set; }   // số nhát Thần Hỏa còn lại
    }

    public static SlotState Apply(SlotState cur, Outcome o)
    {
        var n = new SlotState
        {
            L = o switch
            {
                Outcome.Great => Math.Min(MaxLevel, cur.L + 2),
                Outcome.Success => Math.Min(MaxLevel, cur.L + 1),
                Outcome.Damage => Math.Max(0, cur.L - 1),
                Outcome.Break => 0,
                _ => cur.L,
            },
            C = cur.C, F = cur.F,
        };
        if (cur.F > 0) n.F = cur.F - 1;
        else if (++n.C >= FrenzyCharge) { n.C = 0; n.F = FrenzyStrikes; }
        if (o == Outcome.Break) { n.C = 0; n.F = 0; }
        return n;
    }

    /// <summary>% cộng thêm vào Công / Sinh lực / Phòng của món đang mặc ở ô.</summary>
    public static double BonusPct(int level) => 3.0 * level + 0.2 * level * level;   // cân bằng 11/2026
    public static long StoneCost(ItemDef item, int level) => Math.Max(50L, GameData.SellPrice(item) * 2L) * (level + 1);
    public static int IronCost(int level) => level >= 12 ? 3 : level >= 8 ? 2 : 1;

    // ---- lưu trữ ----
    public sealed record UndoState(string Slot, SlotState Before);
    public sealed record LogLine(int N, string Slot, string Item, int From, int To, string Kind, long At);
    public sealed class ForgeSave
    {
        public Dictionary<string, SlotState> Slots { get; set; } = new();
        public int N { get; set; }
        public UndoState? Undo { get; set; }
        public List<LogLine> Log { get; set; } = new();
    }

    public static Dictionary<string, int> Levels(Player p) =>
        (p.ForgeLevels ?? "").Split(';', StringSplitOptions.RemoveEmptyEntries).Select(x => x.Split(':'))
            .Where(x => x.Length == 2 && int.TryParse(x[1], NumberStyles.Integer, CultureInfo.InvariantCulture, out _))
            .ToDictionary(x => x[0], x => Math.Clamp(int.Parse(x[1], CultureInfo.InvariantCulture), 0, MaxLevel));

    public static ForgeSave Load(Player p)
    {
        if (string.IsNullOrWhiteSpace(p.ForgeState)) return new();
        try { return JsonSerializer.Deserialize<ForgeSave>(p.ForgeState) ?? new(); }
        catch (JsonException) { return new(); }
    }

    public static void Store(Player p, ForgeSave s)
    {
        if (s.Log.Count > HistoryKeep) s.Log.RemoveRange(HistoryKeep, s.Log.Count - HistoryKeep);
        p.ForgeState = JsonSerializer.Serialize(s);
        p.ForgeLevels = string.Join(';', s.Slots.Where(kv => kv.Value.L > 0)
            .Select(kv => $"{kv.Key}:{kv.Value.L.ToString(CultureInfo.InvariantCulture)}"));
    }

    // ---- vật phẩm ----
    public static readonly ItemDef[] Items =
    [
        new ItemDef(Iron, "Tinh Thiết", "ticket", "#b8c2c4", "Thỏi sắt trăm lần tôi luyện, vân kim loại ánh lên khi gặp lửa.",
            Rarity: "uncommon", Kind: "Nguyên liệu rèn",
            Use: "Dùng ở Luyện chế → Lò Rèn: mỗi nhát rèn tốn 1–3 Tinh Thiết tùy cấp. Mua ở Chợ → Tạp Hóa bà Dĩm."),
        new ItemDef(Rewind, "Hồi Thiên Phù", "ticket", "#e6c46a", "Lá phù vẽ ngược dòng thời gian, đốt lên là nhát búa vừa rồi như chưa từng giáng xuống.",
            Rarity: "epic", Kind: "Phù",
            Use: "Ở Lò Rèn: hoàn nhát rèn gần nhất, kể cả khi vỡ nát. Không hoàn linh thạch, Tinh Thiết. Mua ở Chợ → Tạp Hóa bà Dĩm."),
    ];
}
```

using VanDao.Data;
namespace VanDao.Game;

/// <summary>Lò Rèn: rèn ô trang bị đang mặc. Server roll kết quả; client chỉ hiển thị (S.forge).</summary>
public partial class GameService
{
    private static ItemDef ForgeItem(Player p, string slot)
    {
        if (!GameData.Realms[p.Realm].EquipSlots.Contains(slot)) throw new GameException("Ô trang bị chưa mở.");
        return EquipmentEffects.ValidItems(p).FirstOrDefault(i => i.Slot == slot)
            ?? throw new GameException("Ô này chưa mặc trang bị, không có gì để rèn.");
    }

    public (string msg, string kind) ForgeStrike(Player p, string slot, Random? random = null)
    {
        var item = ForgeItem(p, slot);
        var save = ForgeData.Load(p);
        var cur = save.Slots.GetValueOrDefault(slot) ?? new ForgeData.SlotState();
        if (cur.L >= ForgeData.MaxLevel) throw new GameException($"Đã đạt cực hạn +{ForgeData.MaxLevel}.");
        var stone = ForgeData.StoneCost(item, cur.L);
        var iron = ForgeData.IronCost(cur.L);
        if (p.Count("linh_thach") < stone) throw new GameException($"Cần {stone:N0} linh thạch cho nhát rèn này.");
        if (p.Count(ForgeData.Iron) < iron) throw new GameException($"Cần {iron} Tinh Thiết cho nhát rèn này.");
        p.Add("linh_thach", -stone);
        p.Add(ForgeData.Iron, -iron);

        var outcome = ForgeData.Roll(ForgeData.OddsAt(cur.L, cur.F > 0), (random ?? Random.Shared).NextDouble());
        var next = ForgeData.Apply(cur, outcome);
        save.Undo = new(slot, cur);
        save.Slots[slot] = next;
        save.N++;
        save.Log.Insert(0, new(save.N, slot, item.Id, cur.L, next.L, ForgeData.Key(outcome), DateTimeOffset.UtcNow.ToUnixTimeSeconds()));
        ForgeData.Store(p, save);

        var slotName = GemData.SlotNames.GetValueOrDefault(slot, slot);
        var lit = cur.F == 0 && next.F > 0 ? " Thần Hỏa bùng lên: 3 nhát tới không thể vỡ." : "";
        return outcome switch
        {
            ForgeData.Outcome.Break => Log(p, $"Lò Rèn: {slotName} +{cur.L} vỡ nát, về +0.", "bad"),
            ForgeData.Outcome.Great when next.L >= 10 => Log(p, $"Lò Rèn: Đại thành! {slotName} +{cur.L} → +{next.L}.", "gold"),
            ForgeData.Outcome.Success when next.L >= 10 => Log(p, $"Lò Rèn: {slotName} lên +{next.L}.", "gold"),
            ForgeData.Outcome.Great => ($"Đại thành! {slotName} +{cur.L} → +{next.L}.{lit}", "gold"),
            ForgeData.Outcome.Success => ($"Thành công: {slotName} +{next.L}.{lit}", "good"),
            ForgeData.Outcome.Damage => ($"Tổn hại: {slotName} +{cur.L} → +{next.L}.{lit}", "bad"),
            _ => ($"Bất biến: {slotName} giữ +{cur.L}.{lit}", ""),
        };
    }

    public (string msg, string kind) ForgeUndo(Player p)
    {
        var save = ForgeData.Load(p);
        if (save.Undo is not { } undo) throw new GameException("Không có nhát rèn nào để hoàn.");
        if (p.Count(ForgeData.Rewind) < 1) throw new GameException("Cần 1 Hồi Thiên Phù.");
        p.Add(ForgeData.Rewind, -1);
        var now = save.Slots.GetValueOrDefault(undo.Slot)?.L ?? 0;
        save.Slots[undo.Slot] = undo.Before;
        save.Undo = null;
        var itemId = p.Equipment.FirstOrDefault(e => e.Slot == undo.Slot)?.ItemId ?? "";
        save.Log.Insert(0, new(save.N, undo.Slot, itemId, now, undo.Before.L, "undo", DateTimeOffset.UtcNow.ToUnixTimeSeconds()));
        ForgeData.Store(p, save);
        var slotName = GemData.SlotNames.GetValueOrDefault(undo.Slot, undo.Slot);
        return Log(p, $"Hồi Thiên Phù: {slotName} trở về +{undo.Before.L}.", "good");
    }

    public ForgeDto ForgeStateFor(Player p)
    {
        var save = ForgeData.Load(p);
        var items = EquipmentEffects.ValidItems(p).ToDictionary(i => i.Slot!);
        var slots = GameData.Realms[p.Realm].EquipSlots.Select(slot =>
        {
            var s = save.Slots.GetValueOrDefault(slot) ?? new ForgeData.SlotState();
            var item = items.GetValueOrDefault(slot);
            var max = s.L >= ForgeData.MaxLevel;
            return new ForgeSlotDto(slot, item?.Id, s.L, s.C, s.F,
                ForgeData.BonusPct(s.L), ForgeData.BonusPct(Math.Min(ForgeData.MaxLevel, s.L + 1)),
                item is null || max ? 0 : ForgeData.StoneCost(item, s.L), max ? 0 : ForgeData.IronCost(s.L),
                ForgeData.OddsAt(s.L, s.F > 0).ToArray());
        }).ToList();
        var log = save.Log.Select(l => new ForgeLogDto(l.N, l.Slot, l.Item, l.From, l.To, l.Kind, l.At)).ToList();
        return new ForgeDto(slots, save.Undo is not null, log, ForgeData.MaxLevel, ForgeData.FrenzyCharge);
    }
}

function openForgeLog(){
  let dlg = $('forgeLog');
  if (!dlg){ dlg = document.createElement('dialog'); dlg.id = 'forgeLog'; dlg.className = 'fg-dialog'; document.body.appendChild(dlg); }
  const log = S.forge.log, counts = forgeLogCounts(log);
  const hits = log.filter(e => e.kind !== 'undo'), ups = hits.filter(e => e.kind === 'great' || e.kind === 'success').length;
  dlg.innerHTML = `<header><div><h3>Nhật ký lò rèn</h3><p>${hits.length ? `Đã rèn ${hits.length} nhát · lên cấp ${Math.round(ups / hits.length * 100)}% · vỡ ${counts.break} lần` : 'Chưa có nhát rèn nào.'}</p></div>
      <button type="button" data-fg="log-close" aria-label="Đóng nhật ký">×</button></header>

      `<button type="button" class="k-${k.kind}" data-fg-filter="${k.kind}" aria-pressed="${forgeLogKind === k.kind}">${k.label} <em>${counts[k.kind] || 0}</em></button>`).join('')}</div>
    <div class="fg-list">${forgeLogFilter(log, forgeLogKind).map(e => `<div class="fg-log-row k-${e.kind}"><span>#${e.n}</span><span>${forgeSlotName(e.slot)} · ${ITEM[e.itemId]?.name || ''}</span><b>${e.kind === 'break' ? `+${e.from} · Vỡ nát` : `+${e.from} → +${e.to} · ${forgeKindLabel(e.kind)}`}</b></div>`).join('') || '<p>Không có nhát rèn nào khớp bộ lọc.</p>'}</div>`;
  if (!dlg.open) dlg.showModal();
}

document.addEventListener('click', async e => {
  const slot = e.target.closest('[data-fg-slot]');
  if (slot){ forgeSlot = slot.dataset.fgSlot; forgeFlash = null; renderForge(); return; }
  const filter = e.target.closest('[data-fg-filter]');
  if (filter){ forgeLogKind = filter.dataset.fgFilter; openForgeLog(); return; }
  const b = e.target.closest('[data-fg]'); if (!b) return;
  if (b.dataset.fg === 'log') return openForgeLog();
  if (b.dataset.fg === 'log-close') return $('forgeLog')?.close();
  if (b.dataset.fg === 'strike' && await act(`forge/strike/${forgeSlot}`)){
    forgeFlash = S.forge.log[0]?.kind || null;   // sáng dòng tỉ lệ vừa trúng rồi mờ dần (CSS transition)
    renderForge();
    clearTimeout(forgeFlashTimer);
    forgeFlashTimer = setTimeout(() => { forgeFlash = null; document.querySelectorAll('.fg-odd.flash').forEach(x => x.classList.remove('flash')); }, 650);
  }
  if (b.dataset.fg === 'undo') await act('forge/undo');
});
